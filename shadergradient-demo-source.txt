import React,{useEffect,useRef,useState} from 'react';
import {createRoot} from 'react-dom/client';
import {ShaderGradientCanvas,ShaderGradient,presets} from '@shadergradient/react';

const query=new URLSearchParams(location.search),embedded=query.has('embedded');document.documentElement.classList.toggle('gallery',query.has('gallery'));
const chapterNames=['Surface','Window','Atmosphere','Volume'],chapterStops=[0,.32,.59,.94];
const clamp=(x,a=0,b=1)=>Math.max(a,Math.min(b,x));
const lerp=(a,b,t)=>a+(b-a)*t;
const smooth=(a,b,p)=>{const x=clamp((p-a)/(b-a));return x*x*(3-2*x);};
const rgb=hex=>hex.slice(1).match(/../g).map(n=>parseInt(n,16));
const mixColour=(a,b,t)=>'#'+rgb(a).map((v,i)=>Math.round(lerp(v,rgb(b)[i],t)).toString(16).padStart(2,'0')).join('');
const frames=[{p:0,x:0,y:0,w:100,h:100,r:0,angle:0},{p:.15,x:0,y:0,w:100,h:100,r:0,angle:0},{p:.3,x:43,y:23,w:52,h:54,r:14,angle:-2},{p:.4,x:43,y:23,w:52,h:54,r:14,angle:0},{p:.57,x:37,y:8,w:60,h:84,r:70,angle:3},{p:.68,x:37,y:8,w:60,h:84,r:70,angle:0},{p:.83,x:37,y:8,w:61,h:84,r:0,angle:0},{p:1,x:37,y:8,w:61,h:84,r:0,angle:0}];
function frameAt(p){const index=frames.findIndex(f=>f.p>=p);const b=frames[Math.max(1,index)],a=frames[Math.max(0,index-1)],t=smooth(a.p,b.p,p);return Object.fromEntries(['x','y','w','h','r','angle'].map(k=>[k,lerp(a[k],b[k],t)]));}
function shaderAt(p){const mint=smooth(.38,.59,p),volume=smooth(.7,.9,p),base=presets.halo.props,water=presets.mint.props,sphere=presets.violaOrientalis.props;const numeric=['uAmplitude','uDensity','uSpeed','uStrength','uFrequency','brightness','cAzimuthAngle','cDistance','cPolarAngle','positionX','positionY','positionZ','rotationX','rotationY','rotationZ'];const props={...base,control:'props',lightType:'3d',enableTransition:false,smoothTime:.08,toggleAxis:false};for(const k of numeric)props[k]=lerp(lerp(Number(base[k]||0),Number(water[k]||0),mint),Number(sphere[k]||0),volume);props.color1=mixColour(mixColour('#ab8fba','#96b296',mint),'#b6a5c4',volume);props.color2=mixColour(mixColour('#d7b497','#bdd0a7',mint),'#cfac8c',volume);props.color3=mixColour(mixColour('#d9dfc7','#e9e9d9',mint),'#738969',volume);props.type=p<.46?'plane':p<.74?'waterPlane':'sphere';props.cameraZoom=p<.74?1:1.6;return props;}
const text=[{eyebrow:'01 / A LIVING SURFACE',lines:['An atmosphere','in motion.'],body:'A live 3D gradient, choreographed by the same scroll that moves the page.'},{eyebrow:'02 / A DIFFERENT FRAME',lines:['One surface.','Many possibilities.'],body:'The full-screen scene becomes a browser window. Text, scale and camera travel together.'},{eyebrow:'03 / CHANGE THE MOOD',lines:['Colour changes','the atmosphere.'],body:'Warm light gives way to a softer palette. Shape, camera and composition continue to evolve.'},{eyebrow:'04 / REVEAL THE VOLUME',lines:['There is more','beneath the surface.'],body:'The gradient becomes a dimensional object. Orbit the camera to see its changing depth.'}];
function WordLines({lines,amount,reduced}){let i=0;return <h1>{lines.map((line,row)=><span className="headline-line" key={row}>{line.split(' ').map(word=>{const n=i++,a=reduced?1:smooth(n*.07,n*.07+.42,amount);return <span className="headline-word" key={n} style={{opacity:a,transform:`translateY(${(1-a)*32}px)`,filter:`blur(${(1-a)*5}px)`}}>{word} </span>;})}</span>)}</h1>;}
class RenderBoundary extends React.Component{state={failed:false};static getDerivedStateFromError(){return{failed:true};}render(){return this.state.failed?<div className="renderer-fallback"><b>WebGL rendering is unavailable.</b><p>The scroll layout still works. Try a compatible browser to view the actual shader.</p></div>:this.props.children;}}
function App(){
 const story=useRef(),progressRef=useRef(0),[progress,setProgress]=useState(0),[reduced,setReduced]=useState(query.get('paused')==='1'||matchMedia('(prefers-reduced-motion: reduce)').matches),[motionPaused,setMotionPaused]=useState(false),[manual,setManual]=useState(0),[orbit,setOrbit]=useState(false),[grain,setGrain]=useState(false),[visible,setVisible]=useState(true);
 useEffect(()=>{const observer=new IntersectionObserver(entries=>setVisible(entries[0].isIntersecting),{rootMargin:'80px'});observer.observe(story.current);return()=>observer.disconnect();},[]);
 useEffect(()=>{if(reduced){progressRef.current=manual;setProgress(manual);scrollTo({top:0,behavior:'instant'});return;}let frame,last=0;const tick=time=>{const el=story.current;if(el){const target=clamp((scrollY-el.offsetTop)/(el.offsetHeight-innerHeight));progressRef.current=lerp(progressRef.current,target,.18);if(Math.abs(progressRef.current-target)<.0003)progressRef.current=target;if(time-last>40){setProgress(progressRef.current);last=time;}}frame=requestAnimationFrame(tick);};frame=requestAnimationFrame(tick);return()=>cancelAnimationFrame(frame);},[reduced,manual]);
 useEffect(()=>{const handler=e=>{if(e.source===parent&&e.origin===location.origin&&e.data?.type==='atlas-motion'){setMotionPaused(Boolean(e.data.paused));setReduced(Boolean(e.data.paused));}};addEventListener('message',handler);return()=>removeEventListener('message',handler);},[]);
 const p=clamp(progress),frame=frameAt(p),shader=shaderAt(p),chapter=p<.2?0:p<.45?1:p<.72?2:3;
 const select=i=>{const at=chapterStops[i];if(reduced){setManual(at);return;}const el=story.current;scrollTo({top:el.offsetTop+at*(el.offsetHeight-innerHeight),behavior:'smooth'});};
 const alpha=[1-smooth(.12,.2,p),smooth(.17,.24,p)*(1-smooth(.39,.46,p)),smooth(.43,.5,p)*(1-smooth(.67,.74,p)),smooth(.72,.81,p)];
 const reveal=[clamp(p/.12+.65),clamp((p-.18)/.13),clamp((p-.44)/.13),clamp((p-.73)/.13)];
 const light=p>.46&&p<.73;
 return <><div className={`journey ${reduced?'is-reduced':''}`} ref={story}><section className={`pinned-stage ${light?'light-stage':''}`} aria-label="Scroll-driven Shader Gradient scene" data-chapter={chapterNames[chapter]} data-progress={p.toFixed(3)}><div className="stage-base" style={{background:mixColour('#eef0e5','#e4ead9',smooth(.4,.5,p)*(1-smooth(.67,.77,p)))}}/><div className="gradient-frame" style={{left:frame.x+'%',top:frame.y+'%',width:frame.w+'%',height:frame.h+'%',borderRadius:frame.r,transform:`rotate(${frame.angle}deg)`,boxShadow:p>.2&&p<.7?'0 25px 100px #0003':'none'}}><div className="gradient-canvas">{visible&&<RenderBoundary><ShaderGradientCanvas style={{position:'absolute',inset:0}} lazyLoad={false} pixelDensity={embedded?1:1.25} fov={45} pointerEvents="none"><ShaderGradient {...shader} animate={motionPaused||reduced?'off':'on'} grain={grain&&chapter<3?'on':'off'} cAzimuthAngle={shader.cAzimuthAngle+(orbit&&chapter===3?110:0)}/></ShaderGradientCanvas></RenderBoundary>}</div><div className="browser-chrome" style={{opacity:smooth(.17,.28,p)*(1-smooth(.4,.48,p))}}><span>● ● ●</span><span>gradient.studio / collection</span><span>↗</span></div><div className="window-copy" style={{opacity:smooth(.24,.32,p)*(1-smooth(.38,.45,p))}}><small>GRADIENT STUDIO</small><h2>A little colour.<br/>A lot of possibility.</h2><span>EXPLORE THE COLLECTION →</span></div></div><div className="surface-shade" style={{opacity:alpha[0]*.55}}/>
 <header className="journey-header">{!embedded&&<a href="index.html#explore">← UI Atlas</a>}<span>ACTUAL SHADER GRADIENT / ORIGINAL SCROLL SYSTEM</span><button aria-pressed={reduced} onClick={()=>{setManual(chapterStops[chapter]);setReduced(!reduced);}}>{reduced?'Enable scroll motion':'Reduce motion'}</button></header>
 {text.map((t,i)=><div className={`chapter-copy chapter-${i}`} key={i} style={{opacity:alpha[i],pointerEvents:alpha[i]>.5?'auto':'none'}} aria-hidden={alpha[i]<.5}><small>{t.eyebrow}</small><WordLines lines={t.lines} amount={reveal[i]} reduced={reduced}/><p>{t.body}</p>{i===2&&<div className="parameter-panel"><small>CHOREOGRAPHED PARAMETERS</small><dl><dt>Camera distance</dt><dd>{shader.cDistance.toFixed(2)}</dd><dt>Noise strength</dt><dd>{shader.uStrength.toFixed(2)}</dd><dt>Rotation</dt><dd>{shader.rotationZ.toFixed(0)}°</dd></dl><div className="swatches">{[shader.color1,shader.color2,shader.color3].map((c,i)=><i key={i} style={{background:c}}/>)}</div></div>}{i===0&&<span className="scroll-cue">SCROLL TO EXPLORE ↓</span>}{i===3&&<div className="volume-tools"><button tabIndex={chapter===3?0:-1} aria-pressed={orbit} onClick={()=>setOrbit(!orbit)}>{orbit?'Reset orbit':'Orbit the volume'}</button></div>}</div>)}


 <div className="journey-controls"><nav aria-label="Gradient chapters">{chapterNames.map((name,i)=><button key={name} aria-current={chapter===i?'step':undefined} onClick={()=>select(i)}>{String(i+1).padStart(2,'0')} <span>{name}</span></button>)}</nav><div className="gradient-options"><button aria-pressed={motionPaused} onClick={()=>setMotionPaused(!motionPaused)}>{motionPaused?'Play gradient':'Pause gradient'}</button>{chapter<3&&<button aria-pressed={grain} onClick={()=>setGrain(!grain)}>{grain?'Grain on':'Grain off'}</button>}</div><div className="scroll-progress"><i style={{width:p*100+'%'}}/></div></div>
 </section></div>{!embedded&&<footer><div><small>ONE CONTINUOUS TIMELINE</small><h2>Scroll is the conductor.</h2><p>Actual Shader Gradient rendering, with original layout and text choreography. Colours, camera, scene framing and word reveals share one scroll progress value.</p></div><div><a href="https://shadergradient.co/" target="_blank" rel="noopener">Source inspiration ↗</a><a href="SHADERGRADIENT_NOTICES.txt">Renderer & dependency notices</a><a href="shadergradient-demo-source.txt">Demo implementation ↗</a></div></footer>}</>;
}
createRoot(document.getElementById('root')).render(<App/>);
